local frameCount = 0
local lastCheck = tick()
RunService.RenderStepped:Connect(function()
    frameCount = frameCount + 1
    if tick() - lastCheck >= 1 then
        local fps = math.floor(frameCount / (tick() - lastCheck))
        local ping = math.floor(Stats.Network.ServerStatsItem["Data Ping"]:GetValue())
        StatsLabel.Text = string.format("FPS: %d | Ping: %d ms", fps, ping)
        frameCount = 0
        lastCheck = tick()
    end
end)

local Flags = {
    AntiAFKJump = false,
    FarmSafePlatform = false,
    AutoSummer = false,
    AntiSpectate = false,
    FixLag = false
}

local function CreateToggleButton(name, pos, callback)
    local Btn = Instance.new("TextButton", ScrollFrame)
    Btn.Size = UDim2.new(0.9, 0, 0, IsMobile and 28 or 30)
    Btn.Position = UDim2.new(0.05, 0, 0, pos)
    Btn.BackgroundColor3 = Color3.fromRGB(35, 35, 35)
    Btn.BackgroundTransparency = 0.2
    Btn.Text = name .. ": OFF"
    Btn.TextColor3 = Color3.fromRGB(255, 100, 100)
    Btn.Font = Enum.Font.SourceSansBold
    Btn.TextSize = IsMobile and 11 or 12
    Btn.ZIndex = 3
    
    local enabled = false
    Btn.MouseButton1Click:Connect(function()
        enabled = not enabled
        Btn.Text = name .. ": " .. (enabled and "ON" or "OFF")
        Btn.TextColor3 = enabled and Color3.fromRGB(100, 255, 100) or Color3.fromRGB(255, 100, 100)
        callback(enabled)
    end)
end

-- 1. Anti-AFK Nhảy 1s/Lần
CreateToggleButton("Anti-AFK (Nhảy 1s/Lần)", 5, function(val)
    Flags.AntiAFKJump = val
    if Flags.AntiAFKJump then
        task.spawn(function()
            while Flags.AntiAFKJump do
                task.wait(1)
                pcall(function()
                    local char = LocalPlayer.Character
                    local hum = char and char:FindFirstChildOfClass("Humanoid")
                    if hum and hum.Health > 0 then
                        hum.Jump = true
                    end
                end)
            end
        end)
    end
end)

-- 2. Safe Platform Farm Level
local SafePlatform = nil
CreateToggleButton("Farm Level (Tạo Sàn Safezone)", 40, function(val)
    Flags.FarmSafePlatform = val
    if Flags.FarmSafePlatform then
        pcall(function()
            if not SafePlatform or not SafePlatform.Parent then
                SafePlatform = Instance.new("Part")
                SafePlatform.Name = "EvadeSafePlatform"
                SafePlatform.Size = Vector3.new(35, 2, 35)
                SafePlatform.Position = Vector3.new(0, 3000, 0)
                SafePlatform.Anchored = true
                SafePlatform.Material = Enum.Material.SmoothPlastic
                SafePlatform.Color = Color3.fromRGB(40, 40, 40)
                SafePlatform.Parent = workspace
            end
        end)

        task.spawn(function()
            while Flags.FarmSafePlatform do
                task.wait(0.3)
                pcall(function()
                    local char = LocalPlayer.Character
                    local hrp = char and char:FindFirstChild("HumanoidRootPart")
                    if hrp then
                        hrp.CFrame = CFrame.new(0, 3004, 0)
                    end
                end)
            end
        end)
    else
        pcall(function()
            if SafePlatform then
                SafePlatform:Destroy()
                SafePlatform = nil
            end
        end)
    end
end)

-- 3. Auto Event Summer
CreateToggleButton("Auto Event Summer", 75, function(val)
    Flags.AutoSummer = val
    if Flags.AutoSummer then
        task.spawn(function()
            while Flags.AutoSummer do
                task.wait(0.4)
                pcall(function()
                    local char = LocalPlayer.Character
                    local hrp = char and char:FindFirstChild("HumanoidRootPart")
                    if hrp then
                        for _, item in pairs(workspace:GetDescendants()) do
                            if not Flags.AutoSummer then break end
                            if item:IsA("BasePart") or item:IsA("MeshPart") then
                                local name = item.Name:lower()
                                if name:find("ticket") or name:find("summer") or name:find("token") or name:find("collect") or name:find("chocolate") or name:find("event") then
                                    hrp.CFrame = item.CFrame + Vector3.new(0, 2, 0)
                                    local prompt = item:FindFirstChildOfClass("ProximityPrompt") or item.Parent:FindFirstChildOfClass("ProximityPrompt")
                                    if prompt then
                                        fireproximityprompt(prompt)
                                    end
                                    task.wait(0.2)
                                end
                            end
                        end
                    end
                end)
            end
        end)
    end
end)

-- 4. Ẩn Thân (Anti-Spectate)

    Flags.AntiSpectate = val
    if Flags.AntiSpectate then
        task.spawn(function()
            while Flags.AntiSpectate do
                task.wait(0.3)
                pcall(function()

                    if char then
                        for _, part in pairs(char:GetChildren()) do
                            if part:IsA("BasePart") and part.Name ~= "HumanoidRootPart" then

            end
        end)
    end
end)

-- 5. Fix Lag
CreateToggleButton("Fix Lag (Tối Ưu Cho Mobile)", 145, function(val)
    Flags.FixLag = val
    if Flags.FixLag then
        pcall(function()
            settings().Rendering.QualityLevel = Enum.QualityLevel.Level01
            for _, v in pairs(game:GetDescendants()) do
                if v:IsA("Part") or v:IsA("MeshPart") then
                    v.Material = Enum.Material.SmoothPlastic
                elseif v:IsA("Decal") or v:IsA("Texture") then
                    v:Destroy()
                elseif v:IsA("ParticleEmitter") or v:IsA("Trail") then
                    v.Enabled = false
                end
            end
        end)
    end
end)

local function CreateTextBox(placeholder, pos, onFocusLost)
    local Box = Instance.new("TextBox", ScrollFrame)
    Box.Size = UDim2.new(0.9, 0, 0, IsMobile and 28 or 30)
    Box.Position = UDim2.new(0.05, 0, 0, pos)
    Box.BackgroundColor3 = Color3.fromRGB(25, 25, 25)
    Box.BackgroundTransparency = 0.2
    Box.PlaceholderText = placeholder
    Box.Text = ""
    Box.TextColor3 = Color3.fromRGB(255, 215, 0)
    Box.PlaceholderColor3 = Color3.fromRGB(150, 150, 150)
    Box.Font = Enum.Font.SourceSansBold
    Box.TextSize = IsMobile and 11 or 12
    Box.ZIndex = 3
    
    Box.FocusLost:Connect(function(enterPressed)
        if enterPressed and Box.Text ~= "" then
            onFocusLost(Box.Text)
        end
    end)
end

CreateTextBox("Fake Level...", 180, function(text)
    local num = tonumber(text)
    if num then
        pcall(function()
            if LocalPlayer:FindFirstChild("leaderstats") and LocalPlayer.leaderstats:FindFirstChild("Level") then
                LocalPlayer.leaderstats.Level.Value = num
            end
            local playerGui = LocalPlayer:FindFirstChild("PlayerGui")
            if playerGui then
                for _, v in pairs(playerGui:GetDescendants()) do
                    if v:IsA("TextLabel") and (v.Name:lower():find("level") or v.Text:lower():find("lvl")) then
                        v.Text = "LVL " .. tostring(num)
                    end
                end
            end
        end)
    end
end)

HopBtn.MouseButton1Click:Connect(function()
    HopBtn.Text = "Đang tìm server..."
    pcall(function()
        local Api = "https://games.roblox.com/v1/games/" .. game.PlaceId .. "/servers/Public?sortOrder=Asc&limit=100"
        local Servers = HttpService:JSONDecode(game:HttpGet(Api))
        for _, s in pairs(Servers.data) do
            if s.playing < s.maxPlayers and s.id ~= game.JobId then
                TeleportService:TeleportToPlaceInstance(game.PlaceId, s.id, LocalPlayer)
                break
            end
        end
    end)
end)
